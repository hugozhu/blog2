使用docker容器来实现动态执行AI生成的代码

<!--more-->

```yaml
version: '3.8'

services:
  e2b-sandbox:
    image: e2bdev/code-interpreter:latest
    container_name: e2b-sandbox
    ports:
      - "49999:49999"  # 本地端口映射，供 Python SDK 连接
      - "49983:49983"  # 本地端口映射，供 Python SDK 连接
    environment:
      - TOKEN=dev  # 跳过验证
      - E2B_LOCAL=True
    volumes:
      - ./workspace:/home/user/workspace  # 可选，挂载工作目录
    restart: unless-stopped
```

```python
# pip install e2b-code-interpreter
from e2b_code_interpreter import Sandbox

# Create a E2B Sandbox
with Sandbox(
    debug=True,
    api_key="dev"
) as sandbox:
    # Run code
    sandbox.run_code("x = 1")
    execution = sandbox.run_code("x+=1; x")

    print(execution.text) # outputs 2
```
